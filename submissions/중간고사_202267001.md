"""
문제 1 (1장 + 2장). 역 3km 안 시설 수
=====================================
어느 구청이 지하철역 한 곳에서 직선 3km 안에 있는 어린이집(모의 400곳)을 센다.
역과 시설 좌표는 학번으로 정해지며, 시험용 모의 자료다.

실행: python exam/q1.py --id 학번
수정본은 q1_fixed.py 로 저장해 같은 방식으로 실행한다.
"""

import numpy as np
import geopandas as gpd
from pyproj import Transformer
from shapely.geometry import Point

from common import key, parse_id, rng_of, seed_of

STATIONS = {  # (경도, 위도) 시험용 근사값
    "옥수": (127.0183, 37.5406), "신림": (126.9297, 37.4842),
    "화곡": (126.8400, 37.5410), "잠실": (127.1001, 37.5133),
    "노원": (127.0564, 37.6552), "홍대입구": (126.9249, 37.5572),
}

sid = parse_id()
name = list(STATIONS)[seed_of(sid) % len(STATIONS)]
lon0, lat0 = STATIONS[name]
rng = rng_of(sid, 1)
n = 400
lon = lon0 + rng.uniform(-0.07, 0.07, n)
lat = lat0 + rng.uniform(-0.055, 0.055, n)
fac = gpd.GeoDataFrame({"id": np.arange(n)}, geometry=gpd.points_from_xy(lon, lat),
                       crs="EPSG:4326")
station = Point(lon0, lat0)

print(f"역: {name}")
print(f"[확인 1] 객체 수 {len(fac)}, 좌표계 {fac.crs}, 기하 유형 {fac.geom_type.unique()[0]}")
print(f"[확인 2] 범위(경도) {lon.min():.4f} ~ {lon.max():.4f}, (위도) {lat.min():.4f} ~ {lat.max():.4f}")

# [수정] 경위도(도) 단위 버퍼는 경도 1도가 위도 1도보다 짧아(위도 37.5도에서 약 88km)
# 실제로는 동서로 눌린 타원이 된다. 미터 좌표계(EPSG:5179)로 옮겨 3000m 버퍼를 만든다.
fac_m = fac.to_crs("EPSG:5179")
station_m = gpd.GeoSeries([station], crs="EPSG:4326").to_crs("EPSG:5179").iloc[0]
buf = station_m.buffer(3000)
inside = fac_m[fac_m.within(buf)]
area_km2 = buf.area / 1e6          # m2 -> km2

# 실제 거리 확인용: 두 점을 미터 좌표계로 옮겨 측정한다
to_m = Transformer.from_crs("EPSG:4326", "EPSG:5179", always_xy=True)
sx, sy = to_m.transform(lon0, lat0)
fx, fy = to_m.transform(fac.geometry.x.values, fac.geometry.y.values)
dist_m = np.hypot(fx - sx, fy - sy)
true_ids = set(fac.index[dist_m <= 3000])
missed = len(true_ids - set(inside.index))

print(f"3km 안 시설 수: {len(inside)}곳")
print(f"권역 면적: {area_km2:.2f} km2")
print(f"실제 3km 안인데 빠진 시설: {missed}곳")
key("count_3km", len(inside))
key("area_km2", float(area_km2))
key("missed", missed)

```python
"""
문제 2 (3장). 지도에 0개로 나온 업종을 비었다고 판정해도 되는가
=================================================================
서울 412개 동에서, 상가정보(등록부)에 있는 점포 중 지도(OSM)에 나타난 비율이 포착률이다.
다섯 업종의 포착률과 동별 점포 수는 학번으로 정해지는 모의 자료다.

실행: python exam/q2.py --id 학번
"""

import numpy as np

from common import key, parse_id, rng_of

KINDS = ["서점", "안경", "화장품", "약국", "카페"]
N_DONG = 412

sid = parse_id()
rng = rng_of(sid, 2)
lam = rng.uniform(0.3, 3.5, len(KINDS))
registry = [rng.negative_binomial(1.5, 1.5 / (1.5 + l), N_DONG) for l in lam]  # 동별 등록 점포 수
p_true = rng.uniform(0.04, 0.6, len(KINDS))
over = int(rng.integers(0, len(KINDS)))          # 대응표 문제가 있는 업종 하나
osm_total = []
for k, reg in enumerate(registry):
    r = int(reg.sum())
    osm_total.append(int(r * rng.uniform(1.03, 1.15)) if k == over else int(rng.binomial(r, p_true[k])))

CUT_BIZ, CUT_PUB = 0.80, 0.20

print(f"{'업종':<6}{'등록':>7}{'지도':>7}{'포착률':>8}{'0개 동 비율':>12}{'사후확률':>10}  판정")
for k, name in enumerate(KINDS):
    reg = registry[k]
    p = osm_total[k] / int(reg.sum())
    prior0 = float((reg == 0).mean())
    # [수정 2] 포착률은 확률이라 1을 넘을 수 없다. 1을 넘으면 등록부-지도 대응표가 어긋난 것이므로
    # (1-p)가 음수가 되는 계산을 하지 않고 사후확률은 NA, 판정은 '대응표 점검'으로 뺀다.
    if p > 1:
        posterior = None
        verdict = "대응표 점검"
    else:
        # [수정 1] 0개 동 비율은 사전확률 P(N=0)일 뿐, 지도에 0개로 '보인' 뒤의 확률이 아니다.
        # 3장 8.1절: P(진짜 0개 | 0개로 보임) = P(N=0) / sum_n P(N=n) * (1-p)^n
        n_vals, cnt = np.unique(reg, return_counts=True)
        p_n = cnt / cnt.sum()
        p_obs0 = float(np.sum(p_n * (1 - p) ** n_vals))
        posterior = prior0 / p_obs0
        # [수정 2] 판정을 시험지의 네 가지로: 기업 컷라인 0.80, 공공 컷라인 0.20
        if posterior >= CUT_BIZ:
            verdict = "공백 판정"
        elif posterior >= CUT_PUB:
            verdict = "현장 확인 대상"
        else:
            verdict = "판정 불가"
    post_txt = "NA" if posterior is None else f"{posterior:.3f}"
    print(f"{name:<6}{int(reg.sum()):>7}{osm_total[k]:>7}{p:>8.3f}{prior0:>12.3f}{post_txt:>10}  {verdict}")
    key(f"p_{name}", float(p))
    key(f"post_{name}", "NA" if posterior is None else float(posterior))
    key(f"verdict_{name}", verdict)
```

### 답안

**2-1** (3점) 결함이 있는 줄과, 그 계산이 사후확률이 아닌 이유

>> 답: 원본 35행의 posterior = prior0이 결함이며, prior0은 지도를 관측하기 전 등록부의 0개 동 비율인 사전확률 P(N=0)일 뿐, 지도의 포착률 p와 "지도에 0개로 관측되었다"는 조건을 반영한 베이즈 사후확률 P(N=0 | OSM=0)이 아니다.


**2-2** (6점) 수정 후 표: 업종별 포착률, 사후확률, 판정

>> 답: | 업종    | 포착률 (p) | 사후확률 (post) | 판정 (verdict) |
|-----|---------|------------|----------------|
| 서점    | 1.045      | NA              | 대응표 점검|
| 안경    | 0.489      | 0.763           | 현장 확인 대상 |
| 화장품   | 0.275      | 0.349           | 현장 확인 대상 |
| 약국    | 0.584      | 0.849           | 공백 판정 |
| 카페    | 0.296      | 0.826           | 공백 판정 |


**2-3** (3점) 판정이 수정 전과 달라진 업종 하나를 골라 숫자로 설명

>> 답: 약국의 경우 수정 전에는 포착률을 무시한 사전확률 0.660으로 컷라인 0.80에 못 미쳐 '판정 불가'였으나, 포착률 0.584를 적용해 베이즈 규칙으로 계산한 사후확률은 0.849로 상승하여 기업 컷라인(0.80)을 넘어 '공백 판정'으로 바뀌었다.


**2-4** (3점) 포착률이 1을 넘은 업종을 어떻게 처리했고, 그 업종에 사후확률을 계산하면 안 되는 이유

>> 답: 서점의 포착률이 1.045로 1을 넘었으며, 사후확률을 계산하지 않고 'NA'로 출력하며 판정을 '대응표 점검'으로 분리했다. 포착률은 등록 점포가 지도에 나타날 확률이므로 1을 초과할 수 없으며, 1을 넘는다는 것은 등록부와 지도의 업종 분류 대응표가 어긋났음을 의미하므로 (1-p)가 음수가 되는 왜곡된 수치를 방지하기 위해 계산에서 제외해야 한다.

"""
문제 3 (4장). 새 지역에서도 이 점수가 나오는가
===============================================
500개 관측점(spatial_points.geojson) 가운데 학번으로 400개를 뽑아,
접근성 지표(target)를 예측하는 랜덤 포레스트의 R2를 보고한다.
보고서의 문장: "이 모델은 아직 조사하지 않은 새 지역에서도 이 점수를 낸다."

피처 사전
  x_coord, y_coord     관측점 좌표(m)
  dist_center_0..2     관심 지점 3곳까지 거리(m)
  survey_score         사후 만족도 조사 점수 -- 지표가 확정된 뒤 현장 조사로 얻음

실행: python exam/q3.py --id 학번
"""

import geopandas as gpd
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from sklearn.model_selection import GroupKFold, KFold, cross_val_score

from common import PRACTICE, key, parse_id, rng_of

sid = parse_id()
rng = rng_of(sid, 3)
gdf = gpd.read_file(PRACTICE / "chapter4" / "data" / "spatial_points.geojson")
sub = gdf.iloc[rng.choice(len(gdf), 400, replace=False)].reset_index(drop=True)
sub["survey_score"] = sub["target"].values + rng.normal(0, 3, len(sub))

centers = np.array([[320000, 4165000], [315000, 4155000], [330000, 4170000]])
x, y = sub["x_coord"].values, sub["y_coord"].values
for i, (cx, cy) in enumerate(centers):
    sub[f"dist_center_{i}"] = np.hypot(x - cx, y - cy)

# [수정 1] 목표값 누수: survey_score는 지표(target)가 확정된 뒤 현장 조사로 얻는 값이라
# 새 지역을 예측하는 시점에는 알 수 없다(27행에서 target + 잡음으로 만들어짐). 피처에서 뺀다.
features = ["x_coord", "y_coord", "dist_center_0", "dist_center_1", "dist_center_2"]
X, t = sub[features].values, sub["target"].values
model = RandomForestRegressor(n_estimators=100, max_depth=10, random_state=42)
cv = KFold(n_splits=5, shuffle=True, random_state=42)
scores = cross_val_score(model, X, t, cv=cv, scoring="r2")

# [수정 2] 공간 분할 누수: 무작위 분할은 가까운 관측점을 학습과 검증에 나눠 넣어
# '새 지역'이 아니라 '본 지역 사이 빈 곳 메우기' 성능을 잰다. 4장 7.2절대로
# 좌표를 5km 격자 블록으로 잘라 같은 블록은 같은 폴드에 넣는다(GroupKFold 5).
block_size = 5000
block_id = ((sub.geometry.x // block_size).astype(int) * 10000
            + (sub.geometry.y // block_size).astype(int)).values
block_scores = cross_val_score(model, X, t, cv=GroupKFold(n_splits=5), groups=block_id, scoring="r2")

r2_random = float(scores.mean())
r2_block = float(block_scores.mean())
diff = r2_random - r2_block
block_sd = float(block_scores.std())
# 4장 7절: 재려는 차이가 폴드 간 표준편차보다 작으면 그 차이를 결과로 인용하지 않는다
cite_diff = 1 if diff > block_sd else 0

print(f"관측점 {len(sub)}개, 피처 {features}")
print(f"[무작위 5겹] R2: {np.round(scores, 3).tolist()}  평균 {r2_random:.3f}")
print(f"[5km 블록 {len(np.unique(block_id))}개, GroupKFold 5] R2: {np.round(block_scores, 3).tolist()}  "
      f"평균 {r2_block:.3f}, 폴드 표준편차 {block_sd:.3f}")
print(f"무작위 - 블록 차이 = {diff:.3f} ({'표준편차보다 큼: 인용 가능' if cite_diff else '표준편차 이하: 인용하지 않음'})")
key("r2_random", r2_random)
key("r2_block", r2_block)
key("diff", diff)
key("block_sd", block_sd)
key("cite_diff", cite_diff)
```

"""
문제 4 (5장). CNN을 도입해도 되는가
====================================
토지피복 분류에서 세 방법의 폴드별 정확도(5개 공간 타일 폴드)가 아래처럼 나왔다.
이 값은 시험용 모의 점수이며, 실제 CNN을 학습해 얻은 값이 아니다. 학번마다 다르다.

  R1  고정 임계 규칙  (학습 없음)
  L   로지스틱 회귀  (값싼 기준선)
  CNN 합성곱 신경망  (연 개발·운영비가 L보다 더 든다)

도입 조건(기관이 정한 값)
  연 판독 건수 200,000건, 오분류 1건 처리비 2,000원, CNN 추가비용 연 3,000,000원

보고서의 문장: "평균 정확도가 가장 높은 방법을 도입한다."

실행: python exam/q4.py --id 학번
"""

import numpy as np

from common import key, parse_id, rng_of

ANNUAL_CASES = 200_000
COST_PER_ERROR = 2_000
CNN_EXTRA_COST = 3_000_000

sid = parse_id()
rng = rng_of(sid, 4)
fold_shift = rng.normal(0, 0.012, 5)            # 폴드마다 어려운 정도(모든 방법에 공통)
delta = rng.uniform(-0.005, 0.045)              # CNN이 L보다 낫거나 못한 정도
acc = {
    "R1": 0.588 + fold_shift + rng.normal(0, 0.008, 5),
    "L": 0.947 + fold_shift + rng.normal(0, 0.008, 5),
    "CNN": 0.947 + delta + fold_shift + rng.normal(0, 0.008, 5),
}

print(f"{'방법':<5}{'폴드1':>8}{'폴드2':>8}{'폴드3':>8}{'폴드4':>8}{'폴드5':>8}{'평균':>8}")
for m, a in acc.items():
    print(f"{m:<5}" + "".join(f"{v:>8.3f}" for v in a) + f"{a.mean():>8.3f}")

best = max(acc, key=lambda m: acc[m].mean())
print(f"평균이 가장 높은 방법: {best}")

# [수정] '평균이 가장 높은 방법'은 (1) 폴드마다 차이가 같은 방향인지(5장 7절)와
# (2) 그 차이가 CNN 추가비용을 갚는지(5장 9절)를 보지 않는다. 같은 폴드끼리 짝지어 CNN-L을 비교한다.
d = acc["CNN"] - acc["L"]                                   # 폴드별 차이 (같은 폴드끼리)
n_pos = int((d > 0).sum())                                  # 차이가 양수인 폴드 수
diff_mean = float(d.mean())
breakeven = CNN_EXTRA_COST / (ANNUAL_CASES * COST_PER_ERROR)  # 갚기 위해 필요한 정확도 차이
net_saving_won = int(round(diff_mean * ANNUAL_CASES * COST_PER_ERROR - CNN_EXTRA_COST))
# 다섯 폴드 전부 CNN이 낫고(방향 일치), 평균 차이가 손익분기를 넘어야(연 순이익 > 0) 도입
adopt = (n_pos == 5) and (diff_mean > breakeven)
decision = "CNN 도입" if adopt else "CNN 도입 안 함"

print("CNN - L 폴드별 차이: " + ", ".join(f"{v:+.4f}" for v in d))
print(f"차이가 양수인 폴드: {n_pos}/5, 평균 차이: {diff_mean:+.4f}")
print(f"손익분기 차이: {CNN_EXTRA_COST:,} / ({ANNUAL_CASES:,} x {COST_PER_ERROR:,}) = {breakeven:.4f}")
print(f"연 순이익: {diff_mean:.4f} x {ANNUAL_CASES:,} x {COST_PER_ERROR:,} - {CNN_EXTRA_COST:,} = {net_saving_won:,}원")
print(f"기준선 R1 대비 L의 평균 차이: {float((acc['L'] - acc['R1']).mean()):+.4f}")
print(f"결론: {decision}")
key("best_by_mean", best)
key("diff_mean", diff_mean)
key("n_pos", n_pos)
key("breakeven", float(breakeven))
key("net_saving_won", net_saving_won)
key("decision", decision)
```

```python
"""
문제 5 (6장). 거짓변화 필터는 얼마나 도움이 되는가
====================================================
변화 탐지 모델이 낸 예측 마스크(pred)를 진짜 변화(true_change)와 비교한다.
128x128 래스터 가운데 학번으로 정한 96x96 구역을 쓴다. 픽셀 한 변은 10m이다.
필터는 NDVI 차분 0.25 초과 + 최소 면적 1ha 이상이다.

보고서의 문장: "정확도가 높고 필터를 거친 뒤에도 유지되므로 이 모델은 믿을 만하다."

실행: python exam/q5.py --id 학번
"""

import numpy as np
from scipy import ndimage

from common import PRACTICE, key, parse_id, rng_of

sid = parse_id()
rng = rng_of(sid, 5)
d = np.load(PRACTICE / "chapter6" / "data" / "change_masks.npz")
r0, c0 = (int(v) for v in rng.integers(0, 33, 2))
sl = (slice(r0, r0 + 96), slice(c0, c0 + 96))
ndvi1, ndvi2 = d["ndvi_t1"][sl], d["ndvi_t2"][sl]
true = d["true_change"][sl].astype(bool)
pred = d["pred"][sl].astype(bool)

PIXEL_M = 10.0
# [수정] 면적은 한 변의 제곱이다. 10m x 10m = 100m2 = 0.01ha (원본은 10/10000 = 0.001ha로 계산해
# 1ha 기준이 1000px, 즉 실제 10ha가 되어 진짜 변화 덩어리까지 모두 지워졌다)
PIXEL_HA = PIXEL_M ** 2 / 10000.0     # 픽셀 한 개의 면적(ha)


def score(p, t):
    tp = np.logical_and(p, t).sum()
    fp = np.logical_and(p, ~t).sum()
    fn = np.logical_and(~p, t).sum()
    tn = np.logical_and(~p, ~t).sum()
    acc = (tp + tn) / p.size
    iou = tp / (tp + fp + fn) if (tp + fp + fn) else 0.0
    prec = tp / (tp + fp) if (tp + fp) else 0.0
    rec = tp / (tp + fn) if (tp + fn) else 0.0
    return acc, iou, prec, rec


print(f"구역 시작 위치 ({r0}, {c0}), 진짜 변화 {true.sum()}px, 예측 변화 {pred.sum()}px")
a0, i0, p0, r0_ = score(pred, true)
print(f"필터 전: 정확도 {a0:.3f} | IoU {i0:.3f} | 정밀도 {p0:.3f} | 재현율 {r0_:.3f}")

ok = np.abs(ndvi2 - ndvi1) > 0.25
f1 = np.logical_and(pred, ok)
min_px = int(1.0 / PIXEL_HA)            # 1ha에 해당하는 픽셀 수
lbl, n = ndimage.label(f1)
sizes = ndimage.sum(np.ones_like(lbl), lbl, range(1, n + 1))
keep = [i + 1 for i, s in enumerate(sizes) if s >= min_px]
filt = np.isin(lbl, keep)

a1, i1, p1, r1 = score(filt, true)
print(f"최소 면적 기준: {min_px}px")
print(f"필터 후: 정확도 {a1:.3f} | IoU {i1:.3f} | 정밀도 {p1:.3f} | 재현율 {r1:.3f}")
# [보고 수정] 정확도는 배경(TN)이 대부분을 채워 변화 탐지 성능을 대표하지 못한다(6장 6.4절).
# '전부 배경'이라고만 답해도 나오는 정확도를 기준으로 함께 찍고, 판단은 IoU·정밀도·재현율로 한다.
acc_all_bg = float((~true).sum() / true.size)
print(f"참고: 전부 배경이라고 답할 때의 정확도 {acc_all_bg:.3f} (변화를 하나도 못 잡아도 이 값)")
print(f"보고: 필터 후 IoU {i0:.3f} -> {i1:.3f}, 정밀도 {p0:.3f} -> {p1:.3f}, "
      f"재현율 {r0_:.3f} -> {r1:.3f} (필터는 FP만 지우므로 재현율은 올라가지 않는다)")
key("min_px", min_px)
key("acc_after", float(a1))
key("iou_before", float(i0))
key("iou_after", float(i1))
key("precision_before", float(p0))
key("precision_after", float(p1))
key("recall_before", float(r0_))
key("recall_after", float(r1))
```

### 수정한 코드 전문 (`q6_fixed.py`)

```python
"""
문제 6 (7장). 발주량을 어떻게 정해야 손실이 줄어드는가
========================================================
점포 6곳 x 품목 2종의 일별 수요(store_demand.parquet)에서, 학번으로 점포 하나와 품목 하나를 정한다.
요일별 평균 수요로 예측하고, 예측값을 그대로 발주한다.
결품 1단위 손실 Cu, 잉여 1단위 손실 Co는 7장 8.1절의 가정값이다.

보고서의 문장: "검증용 30%를 무작위로 떼어 평가했으므로 실제 운영에서도 이 손실이 난다."

실행: python exam/q6.py --id 학번
"""

import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split

from common import PRACTICE, key, parse_id, seed_of

COSTS = {"도시락": (1200, 2000), "우산": (15000, 1500)}   # (Cu, Co)

sid = parse_id()
s = seed_of(sid)
df = pd.read_parquet(PRACTICE / "chapter7" / "data" / "store_demand.parquet")
stores = sorted(df["store_id"].unique())
items = sorted(df["item"].unique())
store, item = stores[s % len(stores)], items[(s // len(stores)) % len(items)]
g = df[(df["store_id"] == store) & (df["item"] == item)].sort_values("day").reset_index(drop=True)
Cu, Co = COSTS[item]
cr = Cu / (Cu + Co)

# [수정 1] 무작위 분할은 미래 날짜를 학습에 섞어 과거를 '예측'하게 만든다. 운영에서는 과거로 미래를
# 예측하므로 시간 순서(day로 정렬됨) 앞 70%를 학습, 뒤 30%를 검증에 쓴다.
n_train = int(len(g) * 0.7)
train, test = g.iloc[:n_train], g.iloc[n_train:]
dow_mean = train.groupby("dow")["demand"].mean()
forecast = test["dow"].map(dow_mean).values
actual = test["demand"].values


def daily_cost(order):
    short = np.maximum(actual - order, 0)
    over = np.maximum(order - actual, 0)
    return float((Cu * short + Co * over).mean())


# 비교용: 예측값을 반올림한 점추정 발주 (원본 방식)
order_point = np.round(forecast)
cost_point = daily_cost(order_point)

# [수정 2] 예측값을 그대로 발주하면 결품(Cu)과 잉여(Co) 손실이 같다고 보는 셈이다(임계비 0.5).
# 7장: 학습 구간 잔차(실제 - 요일 평균 예측)의 임계비 분위수만큼 발주를 보정한다.
resid_train = train["demand"].values - train["dow"].map(dow_mean).values
offset = float(np.quantile(resid_train, cr))
order_nv = np.round(forecast + offset)
cost_newsvendor = daily_cost(order_nv)
saving_pct = (cost_point - cost_newsvendor) / cost_point * 100

print(f"점포 {store}, 품목 {item}, Cu={Cu}, Co={Co}, 임계비={cr:.4f}")
print(f"학습 {len(train)}일 / 검증 {len(test)}일 (시간순 분할: 앞 70% 학습)")
print(f"학습 잔차의 {cr:.4f} 분위수(offset): {offset:+.3f}")
print(f"점추정 발주 일평균 손실: {cost_point:,.0f}원")
print(f"임계비 발주 일평균 손실: {cost_newsvendor:,.0f}원")
print(f"절감률: {saving_pct:.2f}%")
key("critical_ratio", float(cr))
key("cost_per_day", int(round(cost_point)))
key("cost_point", int(round(cost_point)))
key("cost_newsvendor", int(round(cost_newsvendor)))
key("offset", offset)
key("saving_pct", float(saving_pct))
```

### 답안

**6-1** (3점) 결함 두 개: 위치와 고치는 방법

>> 답: 위치: 원본 31행 train_test_split(..., shuffle=True), 고치는 방법: 미래 시점으로 과거를 예측하는 왜곡을 막기 위해 날짜순으로 정렬 후 앞 70%를 학습, 뒤 30%를 검증으로 분할한다.
위치: 원본 34행 order = np.round(forecast), 고치는 방법: 결품 손실과 잉여 손실의 불균형을 반영하기 위해 학습 잔차의 임계비 분위수(offset)를 더한 round(forecast + offset)으로 발주량을 산출한다.


**6-2** (6점) 수정 후 표: 임계비, 점추정 손실, newsvendor 손실, 절감률

>> 답: 
| 항목 | 값 |
| --- | --- |
| 임계비 (critical_ratio) | 0.9091 |
| 점추정 손실 (cost_point) | 26,153원 |
| newsvendor(임계비) 발주 손실 (cost_newsvendor) | 15,673원 |
| 절감률 (saving_pct) | 40.07% |
| 발주량 보정값 (offset) | +7.267 |


**6-3** (3점) 임계비가 0.5의 어느 쪽에 있고, 발주량 보정값(offset)의 부호가 왜 그런가

>> 답: 임계비는 0.9091로 0.5보다 큽니다. 우산은 품절 시 결품 손실(15,000원)이 재고 잉여 손실(1,500원)의 10배이다. 따라서 재고를 남기더라도 품절을 방지하기 위해 잔차 분포의 90.9% 분위수인 양수 보정값(+7.267)을 더해 평균 예측치보다 넉넉하게 발주해야 하기 때문이다.


**6-4** (3점) 수정 전 손실(무작위 분할)과 수정 후 점추정 손실(시간순 분할)을 비교하고, 다르다면 이유

>> 답: 수정 전 무작위 분할 손실(33,853원)이 수정 후 시간순 점추정 손실(26,153원)보다 높다. 따라서 검증 구간의 수요 특성을 분석한 결과, 시간순 마지막 30% 기간의 수요 표준편차(4.77)가 무작위 검증 표본(5.36)보다 작아 결품 일수가 적었고, 무작위 분할은 미래 데이터를 혼합하여 운영 현실을 재현하지 못하므로 실제 운영 손실의 근거가 될 수 없다.
